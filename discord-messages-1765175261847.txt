// Importing OpenZeppelin ERC721 contract
import "@openzeppelin/contracts/token/ERC721/ERC721.sol";

// Interface for interacting with a submission contract
interface ISubmission {
    // Struct representing a haiku
    struct Haiku {
        address author; // Address of the haiku author
        string line1;   // First line of the haiku
        string line2;   // Second line of the haiku
        string line3;   // Third line of the haiku
    }

    function mintHaiku(
        string memory _line1,
        string memory _line2,
        string memory _line3
    ) external;

    function counter() external view returns (uint256);

    function shareHaiku(uint256 _id, address _to) external;

    function getMySharedHaikus() external view returns (Haiku[] memory);
}

// Contract for managing Haiku NFTs
contract HaikuNFT is ERC721, ISubmission {
    Haiku[] public haikus; // Array to store haikus
    mapping(address => mapping(uint256 => bool)) public sharedHaikus; 
    uint256 public haikuCounter; 

    constructor() ERC721("HaikuNFT", "HAIKU") {
        haikuCounter = 1; 
    }

    // Function to get the total number of haikus
    function counter() external view override returns (uint256) {
        return haikuCounter;
    }

    // Function to mint a new haiku
    function mintHaiku(
        string memory _line1,
        string memory _line2,
        string memory _line3
    ) external override {
        // Check if the haiku is unique
        string[3] memory haikusStrings = [_line1, _line2, _line3];
        for (uint256 li = 0; li < haikusStrings.length; li++) {
            string memory newLine = haikusStrings[li];
            for (uint256 i = 0; i < haikus.length; i++) {
                Haiku memory existingHaiku = haikus[i];
                string[3] memory existingHaikuStrings = [
                    existingHaiku.line1,
                    existingHaiku.line2,
                    existingHaiku.line3
                ];
                for (uint256 eHsi = 0; eHsi < 3; eHsi++) {
                    string memory existingHaikuString = existingHaikuStrings[eHsi];
                    if (
                        keccak256(abi.encodePacked(existingHaikuString)) ==
                        keccak256(abi.encodePacked(newLine))
                    ) {
                        revert HaikuNotUnique();
                    }
                }
            }
        }

        _safeMint(msg.sender, haikuCounter);
        haikus.push(Haiku(msg.sender, _line1, _line2, _line3));
        haikuCounter++;
    }

    // Function to share a haiku with another address
    function shareHaiku(uint256 _id, address _to) external override {
        require(_id > 0 && _id <= haikuCounter, "Invalid haiku ID");

        Haiku memory haikuToShare = haikus[_id - 1];
        if (haikuToShare.author != msg.sender) {
            revert NotYourHaiku();
        }

        sharedHaikus[_to][_id] = true;
    }

    // Function to get haikus shared with the caller
    function getMySharedHaikus()
        external
        view
        override
        returns (Haiku[] memory)
    {
        uint256 sharedHaikuCount;
        for (uint256 i = 0; i < haikus.length; i++) {
            if (sharedHaikus[msg.sender][i + 1]) {
                sharedHaikuCount++;
            }
        }

        if (sharedHaikuCount == 0) {
            revert NoHaikusShared();
        }

        Haiku[] memory result = new Haiku[](sharedHaikuCount);
        uint256 currentIndex;
        for (uint256 i = 0; i < haikus.length; i++) {
            if (sharedHaikus[msg.sender][i + 1]) {
                result[currentIndex] = haikus[i];
                currentIndex++;
            }
        }

        return result;
    }

    // Custom errors
    error HaikuNotUnique(); 
    error NotYourHaiku();   
    error NoHaikusShared();
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

// Import ERC721 dari OpenZeppelin untuk membuat NFT
import "https://github.com/OpenZeppelin/openzeppelin-contracts/blob/master/contracts/token/ERC721/ERC721.sol";

/// @title Haiku NFT
/// @notice Kontrak ini memungkinkan user membuat NFT berupa haiku,
///         memastikan haiku yang di-mint itu unik, bisa di-share ke alamat lain,
///         dan bisa diambil kembali daftar haiku yang dishare.
contract HaikuNFT is ERC721 {
    // -----------------------------
    // STRUCT & STATE VARIABLES
    // -----------------------------

    /// @notice Struct untuk menyimpan sebuah haiku
    struct Haiku {
        address author; // Penulis haiku
        string line1;   // Baris pertama
        string line2;   // Baris kedua
        string line3;   // Baris ketiga
    }

    /// @notice Array penyimpanan semua haiku yang pernah di-mint
    Haiku[] public haikus;

    /// @notice Mapping untuk menyimpan status share
    /// mapping[receiver][idHaiku] = true jika haiku dengan ID tersebut dishare ke receiver
    mapping(address => mapping(uint256 => bool)) public sharedHaikus;

    /// @notice Counter untuk ID token
    uint256 public haikuCounter;

    // -----------------------------
    // EVENTS
    // -----------------------------
    event HaikuMinted(address indexed author, uint256 indexed haikuId);
    event HaikuShared(address indexed from, address indexed to, uint256 indexed haikuId);

    // -----------------------------
    // CUSTOM ERRORS
    // -----------------------------
    error HaikuNotUnique();   // Jika haiku sudah pernah ada
    error NotYourHaiku();     // Jika mencoba share haiku yang bukan miliknya
    error NoHaikusShared();   // Jika tidak ada haiku yang dishare ke user
    error InvalidHaikuId();   // Jika ID haiku tidak valid

    // -----------------------------
    // CONSTRUCTOR
    // -----------------------------
    constructor() ERC721("HaikuNFT", "HAIKU") {
        // Mulai dari ID = 1 biar gampang dibaca (bukan 0)
        haikuCounter = 1;
    }

    // -----------------------------
    // PUBLIC FUNCTIONS
    // -----------------------------

    /// @notice Fungsi untuk mint haiku baru sebagai NFT
    /// @param _line1 Baris pertama
    /// @param _line2 Baris kedua
    /// @param _line3 Baris ketiga
    function mintHaiku(
        string memory _line1,
        string memory _line2,
        string memory _line3
    ) external {
        // Cek keunikan haiku
        string[3] memory newHaiku = [_line1, _line2, _line3];
        for (uint256 i = 0; i < haikus.length; i++) {
            Haiku memory existing = haikus[i];
            // Bandingkan setiap baris
            if (
                keccak256(abi.encodePacked(existing.line1)) == keccak256(abi.encodePacked(newHaiku[0])) &&
                keccak256(abi.encodePacked(existing.line2)) == keccak256(abi.encodePacked(newHaiku[1])) &&
                keccak256(abi.encodePacked(existing.line3)) == keccak256(abi.encodePacked(newHaiku[2]))
            ) {
                revert HaikuNotUnique();
            }
        }

        // Mint NFT baru
        _safeMint(msg.sender, haikuCounter);

        // Simpan haiku ke array
        haikus.push(Haiku(msg.sender, _line1, _line2, _line3));

        emit HaikuMinted(msg.sender, haikuCounter);

        // Tambah counter
        haikuCounter++;
    }

    /// @notice Fungsi untuk share haiku ke alamat lain
    /// @param _id ID haiku
    /// @param _to Alamat tujuan
    function shareHaiku(uint256 _id, address _to) external {
        if (_id == 0 || _id >= haikuCounter) {
            revert InvalidHaikuId();
        }

        Haiku memory haikuToShare = haikus[_id - 1];

        if (haikuToShare.author != msg.sender) {
            revert NotYourHaiku();
        }

        sharedHaikus[_to][_id] = true;

        emit HaikuShared(msg.sender, _to, _id);
    }
